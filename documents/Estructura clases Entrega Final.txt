
struct Transform
{
    Transform * parent = nullptr;

    glm::vec3 position;
    glm::vec3 rotation;
    glm::vec3 scale;

    glm::mat4 get_transform_matrix () const
    {
        glm::mat4 transform_matrix(1);

        transform_matrix = tranlate (...);
        transform_matrix = rotate (...);
        transform_matrix = scale (...);

        if (parent) return parent->get_transform_matrix () * transform_matrix;

        return transform_matrix;
    }
};

class Node                      // equivalente a GameObject, Actor, Entity

public:

    Transform transform;

    void set_parent (Node & node)

class Model : public Node
{
    shared_ptr<Mesh> mesh;
    shared_ptr<Material> material;

public:

    void render(GLint model_view_matrix_id)
    {
        glm::mat4 transform_matrix = transform.get_transform_matrix();

        material->use ();

        glUniformMatrix4fv(model_view_matrix_id, 1, GL_FALSE, glm::value_ptr(transform_matrix));

        mesh->render ();
    }

};

class Mesh
{
protected:

    enum
    {
        COORDINATES_VBO,
        COLORS_VBO,
        INDICES_EBO,
        VBO_COUNT
    };

    GLuint vbo_ids[VBO_COUNT];      // Ids de los VBOs que se usan
    GLuint vao_id = 0;                  // Id del VAO del cubo

public:

    ~Mesh()
    {
        if (vao_id)
        {
            glDeleteVertexArrays (1, &vao_id);
            glDeleteBuffers (VBO_COUNT, vbo_ids);
        }
    }

    void render ()
    {
        glBindVertexArray (vao_id);
        glDrawElements (GL_TRIANGLES, sizeof(indices), GL_UNSIGNED_INT, 0);
        glBindVertexArray (0);
    }

};

{

        model_view_matrix_id = material->shader_program->get_uniform_location ("model_view_matrix");
        projection_matrix_id = material->shader_program->get_uniform_location ("projection_matrix");

        cube1.material = material;
        cube1.transform.position = vec3(0, 0, -10);

        cube2.mesh = cube_mesh;
        cube2.material = material;
        cube2.transform.position = vec3(1, 0, -10);

        cube2.set_parent (cube1);
    }

    void render ()
    {
        glClear(...);

        cube1.render(model_view_matrix_id);
        cube2.render(model_view_matrix_id);
    }

};
